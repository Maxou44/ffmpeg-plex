# Linux x86_64 - Plex: VAAPI (+libdrm), NVENC/NVDEC (CUDA via clang), OpenCL
--arch=x86_64
--target-os=linux
--enable-cuda-llvm
--enable-libdrm
--disable-beignet
--enable-opencl
--enable-vaapi
--enable-ffnvcodec
--enable-nvdec
--enable-nvenc

--enable-encoder=h264_vaapi,hevc_vaapi,h264_nvenc,hevc_nvenc
--enable-hwaccel=av1_vaapi,av1_nvdec

# hwaccels of the built-in decoders (Plex ships them in the codec packs)
--enable-hwaccel=h264_vaapi,hevc_vaapi,mpeg2_vaapi,vc1_vaapi,wmv3_vaapi,vp8_vaapi,vp9_vaapi
--enable-hwaccel=h264_nvdec,hevc_nvdec,mpeg2_nvdec,vc1_nvdec,wmv3_nvdec,vp8_nvdec,vp9_nvdec
